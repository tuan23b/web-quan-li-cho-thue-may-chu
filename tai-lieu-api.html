Mẫu Code Kết nối & Giao Đồ Tự Động

Mẫu truy vấn SQL lấy đơn hàng chưa kích hoạt và cập nhật trạng thái đã giao vật phẩm 
// 1. Khởi tạo kết nối MySQL Pool có xác thực SSL
const mysql = require('mysql2/promise');

async function processPendingWebstoreDeliveries() {
  const pool = mysql.createPool({
    host: 'mysql-cluster-01.nexusnode.cloud',
    user: 'u1092_webstore',
    password: process.env.NEXUS_DB_PASSWORD,
    database: 'srv_mc_smp_db',
    port: 3306,
    ssl: { rejectUnauthorized: true },
    waitForConnections: true,
    connectionLimit: 10
  });

  // 2. Tìm kiếm danh sách đơn hàng đã thanh toán nhưng chưa kích hoạt
  const [orders] = await pool.query(
    `SELECT id, player_uuid, username, package_rank, coins_amount 
     FROM webstore_donations 
     WHERE status = 'pending' LIMIT 50`
  );

  for (const order of orders) {
    console.log(`[DELIVERY] Trao gói ${order.package_rank} cho ${order.username}`);
    
    // 3. Đánh dấu đã nhận vật phẩm
    await pool.query(
      `UPDATE webstore_donations SET status = 'completed', fulfilled_at = NOW() WHERE id = ?`,
      [order.id]
    );
  }
}
